// Helper function to determine media path
const BASE_URL = 'https://raw.githubusercontent.com/ntshuxekoshirinda/exercises-dataset/main';

function getMediaUrl(filename: string): string {
    if (!filename) return '';
    // If it's a gif, look in videos folder, otherwise images folder
    if (filename.endsWith('.gif')) {
        return `${BASE_URL}/videos/${filename}`;
    }
    return `${BASE_URL}/images/${filename}`;
}

import React from 'react';
import { Pressable, Text, StyleSheet, ViewStyle } from "react-native";
import { Image } from 'expo-image';

interface ExerciseMainProps {
    title: string;
    imageSource: any;
    onPress: () => void;
    style?: ViewStyle;
}

export function ExerciseMain({ title, imageSource, onPress, style }: ExerciseMainProps) {
    return (
        <Pressable onPress={onPress} style={[styles.buttonWrapper, style]}>
            <Image
                source={imageSource}
                style={styles.imageBackground}
                contentFit="cover"
                transition={500} // Fade-in effect
            >
                <View style={styles.overlay}>
                    <Text style={styles.text}>{title}</Text>
                </View>
            </Image>
        </Pressable>
    );
}
